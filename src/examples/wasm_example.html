Text Processing

Enter some text and the Rust code will process it.

Results:

// Rust code: #[wasm_bindgen] pub fn process_text(text: &str) -> String { let word_count = text.split_whitespace().count(); let char_count = text.chars().count(); let uppercase = text.to_uppercase(); format!("Stats: {} words, {} characters\nUppercase: {}", word_count, char_count, uppercase) } // JavaScript approach: function processTextJS() { const text = document.getElementById("text-input").value; const result = wasm.process_text(text); document.getElementById("result").textContent = result; } // Direct DOM manipulation from Rust: #[wasm_bindgen] pub fn update_result() { let input_element = getElementById("text-input"); let input_text = input_element.value(); let result = process_text(&input_text); let result_element = getElementById("result"); result_element.set_innerHTML(&result); }

Fibonacci Sequence

Generate a Fibonacci sequence using Rust.

// Rust code: #[wasm_bindgen] pub fn fibonacci(n: u32) -> u32 { if n <= 1 { return n; } let mut a = 0; let mut b = 1; for _ in 2..=n { let temp = a + b; a = b; b = temp; } b } #[wasm_bindgen] pub fn generate_fibonacci_sequence(count: u32) { let mut sequence = String::from("<ul>"); for i in 0..count { let fib = fibonacci(i); sequence.push_str(&format!("<li>Fibonacci({}) = {}</li>", i, fib)); } sequence.push_str("</ul>"); let sequence_element = getElementById("fibonacci-sequence"); sequence_element.set_innerHTML(&sequence); }
